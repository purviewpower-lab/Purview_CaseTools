# State shared between the UI thread and the background worker runspace
$OutQueue = New-Object 'System.Collections.Concurrent.ConcurrentQueue[object]'
$Shared   = [hashtable]::Synchronized(@{ IsConnected = $false; Account = $null; Tenant = $null; ConnectionId = $null; Progress = $null; `

# Functions used by the actions. They are loaded here and into the worker runspace,
# where every Security & Compliance cmdlet runs so the window stays responsive.
$WorkerFunctions = {
  Function Write-Log {
    param([string]$Message, [string]$Level = 'INFO')
    $line = "[{0}] [{1}] {2}" -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'), $Level, $Message
    $line | Tee-Object -FilePath $logFile -Append | Out-Null
  }

  Function Write-PowerToolsCleanupWarning {
    param([string]$Message)
    try { Write-Log -Message $Message -Level 'WARN' }
    catch { Write-Verbose -Message ('Cleanup diagnostic unavailable: ' + $_.Exception.Message) }
  }
  # Log one outcome and queue it as a row for the active page's results grid
  Function Write-Result {
    param(
      [string]$Case,
      [string]$Member,
      [string]$Action,
      [ValidateSet('Success','Failed','Skipped','Info')][string]$Result,
      [string]$Detail
    )
    $level = if ($Result -eq 'Failed') { 'ERROR' } else { 'INFO' }
    Write-Log ("{0} | case '{1}' | member '{2}' | {3} | {4}" -f $Action, $Case, $Member, $Result, $Detail) $level
    $OutQueue.Enqueue([pscustomobject]@{
      Time   = Get-Date -Format 'HH:mm:ss'
      Case   = $Case
      Member = $Member
      Action = $Action
      Result = $Result
      Detail = $Detail
    })
  }

  # Report how far the current action has got; the status bar shows it as a progress bar
  Function Set-ActionProgress {
    param([int]$Done, [int]$Total, [string]$Unit = 'cases')
    $Shared.Progress = [pscustomobject]@{ Done = $Done; Total = $Total; Unit = $Unit }
  }

  Function Assert-PowerToolsModule {
    if (-not (Get-Module -ListAvailable -Name ExchangeOnlineManagement)) {
      $message = 'ExchangeOnlineManagement Powershell module not detected. Install to use Purview features.'
      Write-Log $message 'ERROR'
      throw $message
    }
  }

  # Connection metadata is diagnostic state, not proof that a token cannot refresh.
  Function Get-IppsConnections {
    if (-not (Get-Command Get-ConnectionInformation -ErrorAction SilentlyContinue)) { return }
    Get-ConnectionInformation -ErrorAction Stop |
      Where-Object { $_.IsEopSession -or ($_.ConnectionUri -match 'compliance') }
  }
  Function Get-IppsConnection {
    $connections = @(Get-IppsConnections)
    if ($Shared.ConnectionId) {
      return ($connections | Where-Object { [string]$_.ConnectionId -eq [string]$Shared.ConnectionId } | Select-Object -First 1)
    }
    # Never adopt an arbitrary connection when several are present.
    if ($connections.Count -eq 1) { return $connections[0] }
    return $null
  }
  Function Test-PowerToolsAuthenticationError {
    param([System.Management.Automation.ErrorRecord]$Record)
    $text = [string]$Record.FullyQualifiedErrorId + ' ' + [string]$Record.Exception.Message
    # Deliberately excludes generic timeouts, forbidden/access-denied and permission errors.
    return ($text -match ('(?i)\b401\b|\bUnauthorized\b|AADSTS(?:50058|50076|50079|50173|70043|700082)|invalid_grant|interactio' + `
'n_required|login_required|token.{0,80}expired|expired.{0,80}token|not connected|no ' + 'active.{0,30}session|session.{0,40}(?:expired|disconnected)'`
))
  }
  Function Connect-Compliance {
    Assert-PowerToolsModule
    # Actions use local scope; keep the dependency available to the persistent worker.
    Import-Module ExchangeOnlineManagement -Global -ErrorAction Stop
    $wasConnected = [bool]$Shared.IsConnected
    $Shared.ReconnectMessage = $null
    $Shared.ReconnectDeadline = $null
    $Shared.ReconnectSecondsRemaining = 0
    if ($wasConnected) {
      $conn = Get-IppsConnection
      $canReport = [bool](Get-Command Get-ConnectionInformation -ErrorAction SilentlyContinue)
      if (-not $canReport -or ($conn -and $conn.State -eq 'Connected')) {
        if ($conn -and $conn.TokenStatus -eq 'Active') { return }
        # Give the module a read-only request before declaring an expired token lost.
        # No mutation is retried by this function.
        try {
          Get-ComplianceCase -CaseType eDiscovery -ErrorAction Stop | Out-Null
          Write-Log 'Existing compliance session passed a read-only check; no interactive reconnection required.'
          return
        } catch {
          if (-not (Test-PowerToolsAuthenticationError -Record $_)) {
            Write-Log ('Compliance read-only check failed without a recognised authentication error: ' + $_.Exception.Message) 'ERROR'
            throw
          }
          Write-Log ('Compliance authentication check failed; requesting a new sign-in: ' + $_.Exception.Message) 'WARN'
        }
      } else {
        Write-Log 'Tracked compliance connection is missing or disconnected; requesting a new sign-in.' 'WARN'
      }
    }
    $Shared.IsConnected = $false
    $Shared.Account = $null
    $Shared.Tenant = $null
    $Shared.ReconnectActive = $true
    $Shared.ReconnectMessage = if ($wasConnected) { 'Reconnecting to Purview. Complete sign-in if prompted.' } else { `
'Connecting to Purview. Complete sign-in if prompted.' }
    try {
      $before = @(Get-IppsConnections)
      $beforeIds = @($before | ForEach-Object { [string]$_.ConnectionId })
      Write-Log $Shared.ReconnectMessage
      # One interactive attempt. Password/MFA entry has no toolbox-imposed deadline.
      # Microsoft authentication and service timeouts still apply.
      Connect-IPPSSession -ErrorAction Stop | Out-Null
      $after = @(Get-IppsConnections)
      $newConnections = @($after | Where-Object { [string]$_.ConnectionId -and $beforeIds -notcontains [string]$_.ConnectionId })
      $conn = $null
      if ($newConnections.Count -eq 1) {
        $conn = $newConnections[0]
      } elseif ($newConnections.Count -gt 1) {
        throw 'Sign-in returned multiple new compliance connections; connection ownership could not be established.'
      } elseif ($Shared.ConnectionId) {
        $conn = $after | Where-Object { [string]$_.ConnectionId -eq [string]$Shared.ConnectionId } | Select-Object -First 1
      }
      if (-not $conn -and $after.Count -eq 1) { $conn = $after[0] }
      if ($conn) {
        Write-Log ('Compliance connection verification: ConnectionId={0}; State={1}; TokenStatus={2}' -f $conn.ConnectionId,$conn.State,`
$conn.TokenStatus)
        if ($conn.State -ne 'Connected' -or $conn.TokenStatus -eq 'Expired') {
          throw 'Sign-in returned, but the identified compliance connection is not healthy.'
        }
        $Shared.ConnectionId = [string]$conn.ConnectionId
        $Shared.Account = [string]$conn.UserPrincipalName
        $Shared.Tenant = [string]$conn.TenantID
      } elseif ($after.Count -gt 0) {
        throw 'Sign-in returned, but the toolbox could not identify its compliance connection among existing connections.'
      } else {
        # Get-ConnectionInformation reports REST connections only. Legacy/module
        # compatibility is verified using a read-only compliance request instead.
        Get-ComplianceCase -CaseType eDiscovery -ErrorAction Stop | Out-Null
        $Shared.ConnectionId = $null
        Write-Log 'No REST compliance metadata available; sign-in verified by a read-only compliance request.' 'WARN'
      }
      if (-not (Get-Command Get-ComplianceCase -ErrorAction SilentlyContinue)) {
        throw 'Sign-in returned, but Get-ComplianceCase is unavailable in the worker runspace.'
      }
      $Shared.IsConnected = $true
      $Shared.ReconnectMessage = if ($wasConnected) { 'Reconnected to Purview successfully' } else { 'Connected to Purview successfully' }
      Write-Log ('{0}. Account={1}; Tenant={2}; ConnectionId={3}' -f $Shared.ReconnectMessage,$Shared.Account,$Shared.Tenant,`
$Shared.ConnectionId)
    } catch {
      $Shared.IsConnected = $false
      $Shared.Account = $null
      $Shared.Tenant = $null
      $Shared.ReconnectMessage = 'Purview connection failed. See the log for details.'
      Write-Log ('Purview connection failed: {0}; ErrorId={1}' -f $_.Exception.Message,$_.FullyQualifiedErrorId) 'ERROR'
      throw
    } finally {
      $Shared.ReconnectActive = $false
      $Shared.ReconnectDeadline = $null
      $Shared.ReconnectSecondsRemaining = 0
    }
  }
  Function Get-CaseMemberEmail {
    [CmdletBinding()]
    param(
      [Parameter(Mandatory=$true)][string]$Case,
      [string]$ExcludeEmail
    )

    try {
      $members = Get-ComplianceCaseMember -Case $Case -ErrorAction Stop

      $emails = $members | ForEach-Object {
        $addr = $_.PrimarySmtpAddress
        if ([string]::IsNullOrWhiteSpace($addr)) { $addr = $_.Name }
        if ($addr) { $addr.Trim() }
      } | Where-Object { $_ } | Select-Object -Unique

      if ($ExcludeEmail) {
        $emails = $emails | Where-Object { $_ -ine $ExcludeEmail }
      }

      return $emails
    }
    catch {
      $msg = "Failed to get members for case '$Case': $($_.Exception.Message)"
      Write-Log $msg 'ERROR'
      throw $msg
    }
  }

  # Case names can contain characters that are not allowed in file names

}
. $WorkerFunctions

# ------------------ Input parsing ------------------
$EmailPattern = '^[^@\s]+@[^@\s]+\.[^@\s]+$'

# Splits a box's text on commas, semicolons and new lines into trimmed, non-empty entries
Function Split-InputList {
  param([string]$Text)
  @(($Text -split "[`,;\r\n]") | ForEach-Object { $_.Trim() } | Where-Object { $_ })
}

# Entries that are not valid email addresses are skipped (the page flags them as you type)
Function ConvertTo-EmailList {
  param([string]$EmailsMultiline)
  Split-InputList $EmailsMultiline |
    Where-Object { $_ -match $EmailPattern } |
    Select-Object -Unique
}

Function ConvertTo-CaseList {
  param([string]$CasesMultiline)
  Split-InputList $CasesMultiline |
    Select-Object -Unique
}

# Resize legacy FolderBrowserDialog windows to a more useful default size.
function Show-PowerToolsFolderPicker {
  param([string]$Description,[string]$SelectedPath='')
  # Use the modern Explorer-style Windows file dialog as a folder selector.
  # A temporary placeholder file name lets the user navigate to a folder and select Open;
  # only the containing folder is returned and no file is created.
  $dlg = New-Object Microsoft.Win32.OpenFileDialog
  $dlg.Title = $Description
  $dlg.CheckFileExists = $false
  $dlg.CheckPathExists = $true
  $dlg.ValidateNames = $false
  $dlg.FileName = 'Select this folder'
  $dlg.Filter = 'Folders|*.folder'
  if ($SelectedPath -and (Test-Path -LiteralPath $SelectedPath -PathType Container)) {
    $dlg.InitialDirectory = [System.IO.Path]::GetFullPath($SelectedPath)
  }
  if ($dlg.ShowDialog($window)) {
    return (Split-Path -Parent $dlg.FileName)
  }
  return $null
}

# Light or Dark, from the Windows "app mode" setting

Function Set-AccelerateChrome {
  param([bool]$On)
  $StatusBar.Resources.Clear()
  if (-not $On) {
    $HeaderRight.Background = $null
    $StatusBar.SetResourceReference([System.Windows.Controls.Border]::BorderBrushProperty, 'CardBorder')
    $StatusBar.BorderThickness = [System.Windows.Thickness]::new(0, 1, 0, 0)
    return
  }
  $toColor = { param($hex) [System.Windows.Media.Color][System.Windows.Media.ColorConverter]::ConvertFromString($hex) }
  $black = [System.Windows.Media.SolidColorBrush]::new((& $toColor '#0A0A0A'))
  $black.Freeze()
  $HeaderRight.Background = $black
  # Status bar text sits on dark green, so these keys are overridden inside the status bar only
  $statusColors = @{
    TextPrimary = '#FFFFFF'; TextSecondary = '#CFE9DA'; Accent = '#6FF08A'; InputBorder = '#1F6B4C'
    SuccessText = '#6FF08A'; DangerText = '#FFA19A'; WarningText = '#FFD166'
  }
  foreach ($key in $statusColors.Keys) {
    # ::new, not New-Object, so the dictionary stores a real Brush
    $brush = [System.Windows.Media.SolidColorBrush]::new((& $toColor $statusColors[$key]))
    $brush.Freeze()
    $StatusBar.Resources[$key] = $brush
  }
  # Seven hard-edged colour bands along the top edge
  $bands = '#45D96B', '#F2E84A', '#7CE8D8', '#3B82F6', '#9E98F5', '#F58CE6', '#F7A85A'
  $rainbow = [System.Windows.Media.LinearGradientBrush]::new()
  $rainbow.StartPoint = [System.Windows.Point]::new(0, 0)
  $rainbow.EndPoint = [System.Windows.Point]::new(1, 0)
  for ($i = 0; $i -lt $bands.Count; $i++) {
    $color = & $toColor $bands[$i]
    $rainbow.GradientStops.Add([System.Windows.Media.GradientStop]::new($color, $i / $bands.Count))
    $rainbow.GradientStops.Add([System.Windows.Media.GradientStop]::new($color, ($i + 1) / $bands.Count))
  }
  $rainbow.Freeze()
  $StatusBar.BorderBrush = $rainbow

Function Set-Theme {
  param([System.Windows.Window]$Target)
  $name = if ($script:ThemeMode -eq 'System') { 'Light' } else { $script:ThemeMode }
  $palette = $Palettes[$name]
  foreach ($key in $palette.Keys) {
    # ::new rather than New-Object: New-Object wraps the brush in a PSObject, and the
    # resource dictionary would store that wrapper, which WPF rejects as a Brush
    $brush = [System.Windows.Media.SolidColorBrush]::new([System.Windows.Media.Color][System.Windows.Media.ColorConverter]::ConvertFromString(`
$palette[$key]))
    $brush.Freeze()
    $Target.Resources[$key] = $brush
  }
  # The slim scroll bars have no arrow buttons; WPF sizes the smallest thumb at half this value
  $Target.Resources[[System.Windows.SystemParameters]::VerticalScrollBarButtonHeightKey] = [double]40
  $Target.Resources[[System.Windows.SystemParameters]::HorizontalScrollBarButtonWidthKey] = [double]40
  if ($Target -eq $window) {
    $isAccelerate = ($name -eq 'Accelerate')
    $accelerateVisibility = if ($isAccelerate) { 'Visible' } else { 'Collapsed' }
    $HeaderNeon.Visibility = if ($name -eq 'Neon') { 'Visible' } else { 'Collapsed' }
    $HeaderTeal.Visibility = if ($name -in 'Neon','Accelerate') { 'Collapsed' } else { 'Visible' }
    $HeaderAccelerate.Visibility = $accelerateVisibility
    $HeaderAccelerateLogo.Visibility = $accelerateVisibility
    $HeaderTitleAccelerate.Visibility = 'Collapsed'
    $HeaderTitle.Visibility = 'Visible'
    # Title pill: shaded in System (Light) and Dark, switched off in Neon and Accelerate.
    # Its colour (#E6061C18, very dark green at 90% opacity) is set on HeaderTitlePill in the XAML.
    if (-not $script:TitlePillBrush) { $script:TitlePillBrush = $HeaderTitlePill.Background }
    if ($name -in 'Neon','Accelerate') {
      $HeaderTitlePill.Background = $null
      $HeaderTitlePill.Padding = New-Object System.Windows.Thickness(8,0,0,0)
    } else {
      $HeaderTitlePill.Background = $script:TitlePillBrush
      $HeaderTitlePill.Padding = New-Object System.Windows.Thickness(16,3,16,5)
    }
    Set-AccelerateChrome -On $isAccelerate
    $LinkLogs.SetResourceReference([System.Windows.Controls.Control]::ForegroundProperty, 'StatusText')
    $ThemeToggle.Content = switch ($script:ThemeMode) { 'Dark' { 'Theme: Dark' } 'Neon' { 'Theme: Neon' } 'Accelerate' { 'Theme: Accelerate' } `
default { 'Theme: System (Light)' } }
  }
}

Function Test-Number($value) {
  $n = 0.0
  [double]::TryParse([string]$value, [System.Globalization.NumberStyles]::Float, [System.Globalization.CultureInfo]::InvariantCulture, [ref]$n) `
-and -not [double]::IsNaN($n) -and -not [double]::IsInfinity($n)
}

# Restores the saved window position only if it is still on one of the screens
Function Restore-PowerToolsSetting {
  param($Settings)
  if (-not $Settings) { return 0 }
  if ($Settings.Theme -in 'Dark','Neon','Accelerate') { $script:ThemeMode = [string]$Settings.Theme } else { $script:ThemeMode = 'System' }
  $screen = New-Object System.Windows.Rect([System.Windows.SystemParameters]::VirtualScreenLeft, `
[System.Windows.SystemParameters]::VirtualScreenTop, [System.Windows.SystemParameters]::VirtualScreenWidth, `
[System.Windows.SystemParameters]::VirtualScreenHeight)
  $haveSize = (Test-Number $Settings.Width) -and (Test-Number $Settings.Height)
  $width  = if ($haveSize) { [Math]::Max($window.MinWidth,  [Math]::Min([double]$Settings.Width,  $screen.Width)) } else { $window.Width }
  $height = if ($haveSize) { [Math]::Max($window.MinHeight, [Math]::Min([double]$Settings.Height, $screen.Height)) } else { $window.Height }
  $restored = $false
  if ((Test-Number $Settings.Left) -and (Test-Number $Settings.Top)) {
    $left = [double]$Settings.Left; $top = [double]$Settings.Top
    $topLeft     = New-Object System.Windows.Point(($left + 20), ($top + 10))
    $bottomRight = New-Object System.Windows.Point(($left + $width - 20), ($top + $height - 10))
    if ($screen.Contains($topLeft) -and $screen.Contains($bottomRight)) {
      $window.WindowStartupLocation = 'Manual'
      $window.Left = $left; $window.Top = $top
      $window.Width = $width; $window.Height = $height
      $restored = $true
    }
  }
  # Otherwise the window opens centred on the main screen, so fit it to that screen
  if ($haveSize -and -not $restored) {
    $workArea = [System.Windows.SystemParameters]::WorkArea
    $window.Width  = [Math]::Max($window.MinWidth,  [Math]::Min([double]$Settings.Width,  $workArea.Width))
    $window.Height = [Math]::Max($window.MinHeight, [Math]::Min([double]$Settings.Height, $workArea.Height))
  }
  if ($Settings.Maximized -eq $true) { $window.WindowState = 'Maximized' }
  $page = 0
  if ((Test-Number $Settings.Page) -and [int]$Settings.Page -ge 0 -and [int]$Settings.Page -le 9) { $page = [int]$Settings.Page }
  $page
}

# Best-effort synchronous diagnostics: no UI pumping, COM calls or background handlers.
# Resource snapshots are process observations, not a diagnosis of a leak/crash.
Function Write-PowerToolsResourceSnapshot {
  param([string]$Stage,[string]$Level='INFO')
  $process=$null
  try{
    $process=[Diagnostics.Process]::GetCurrentProcess();$process.Refresh()
    $elapsed=if($script:SelfTestClock){$script:SelfTestClock.ElapsedMilliseconds}else{0}
    $thread=[Threading.Thread]::CurrentThread
    $line=('RESOURCE | Stage={0}; Test={1}; PID={2}; Bitness={3}; Apartment={4}; Thread={5}; RunElapsedMs={6}; ' + `
'WorkingSetMB={7}; PrivateMB={8}; ManagedMB={9}; Handles={10}; Threads={11}') -f $Stage,$script:SelfTestCaseName,$PID,([IntPtr]::Size*8),`
$thread.GetApartmentState(),$thread.ManagedThreadId,$elapsed,[Math]::Round($process.WorkingSet64/1MB,2),[Math]::Round(`
$process.PrivateMemorySize64/1MB,2),[Math]::Round([GC]::GetTotalMemory($false)/1MB,2),$process.HandleCount,$process.Threads.Count
    Write-Log $line $Level
  }catch{
    # Diagnostics must not alter processing or hide the original exception.
    try{Write-Log ('Resource snapshot unavailable: '+$_.Exception.Message) 'WARN'}catch { Write-PowerToolsCleanupWarning -Message (`
'Best-effort operation failed at line {0}: {1}' -f $_.InvocationInfo.ScriptLineNumber, $_.Exception.Message) }
  }finally{if($process){$process.Dispose()}}
}

Function Get-ResultSummary {
  param($Results)
  $ok      = @($Results | Where-Object { $_.Result -eq 'Success' }).Count
  $failed  = @($Results | Where-Object { $_.Result -eq 'Failed' }).Count
  $skipped = @($Results | Where-Object { $_.Result -eq 'Skipped' }).Count
  $parts = @('{0} succeeded' -f $ok)
  if ($failed)  { $parts += '{0} failed' -f $failed }
  if ($skipped) { $parts += '{0} skipped' -f $skipped }
  [pscustomobject]@{ Text = ($parts -join ', '); Failed = $failed }
}

# ------------------ Background worker ------------------
# One long-lived runspace owns the IPPS connection and runs all cmdlets. The UI thread only
# reads inputs, starts an action, and adds queued results to the grid while the action runs.
$Worker = [runspacefactory]::CreateRunspace($Host)   # share the console host so module warnings still show there
$Worker.ApartmentState = 'STA'          # interactive sign-in needs an STA thread
$Worker.ThreadOptions  = 'ReuseThread'  # keep the connection on the same thread between actions
$Worker.Open()
$Worker.SessionStateProxy.SetVariable('logFile',  $logFile)
$Worker.SessionStateProxy.SetVariable('logDir',   $logDir)
$Worker.SessionStateProxy.SetVariable('OutQueue', $OutQueue)
$Worker.SessionStateProxy.SetVariable('Shared',   $Shared)
$init = [powershell]::Create()
$init.Runspace = $Worker
$null = $init.AddScript("`$ErrorActionPreference = 'Stop'`r`n" + $WorkerFunctions.ToString()).Invoke()
$init.Dispose()

$script:Job = $null

Function Write-PendingOutput {
  param($Results, $Summary)
  $row = $null
  $added = $false
  while ($OutQueue.TryDequeue([ref]$row)) {
    if ($null -ne $Results) { $Results.Add($row); $added = $true }
  }
  if ($added -and $Summary) { $Summary.Text = (Get-ResultSummary $Results).Text }
}

# Polls the running action: streams its results and finishes up when it completes
$PollTimer = New-Object System.Windows.Threading.DispatcherTimer
$PollTimer.Interval = [TimeSpan]::FromMilliseconds(150)
$PollTimer.Add_Tick({
  $job = $script:Job
  if (-not $job) { $PollTimer.Stop(); return }
  $done = $job.Handle.IsCompleted
  Write-PendingOutput $job.Results $job.Summary
  Update-BusyProgress
  if ($Shared.ReconnectActive) { $TxtWorking.Text='Awaiting connection / sign-in...';Set-Status ([string]$Shared.ReconnectMessage) 'Neutral' } `
elseif ($Shared.ReconnectMessage -in @('Reconnected to Purview successfully','Connected to Purview successfully') -and -not $done) { Set-Status (`
[string]$Shared.ReconnectMessage) 'Success';$Shared.ReconnectMessage=$null }
  if (-not $done) { return }

  $PollTimer.Stop()
  $script:Job = $null
  $output = $null
  $failure = $null
  try { $output = $job.PS.EndInvoke($job.Handle) }
  catch {
    $failure = $_.Exception
    if ($failure -is [System.Management.Automation.MethodInvocationException] -and $failure.InnerException) { $failure = $failure.InnerException }
  }
  finally { $job.PS.Dispose() }

  Update-ConnectionText
  Hide-Busy; Set-UIBusy $false

  if ($failure) {
    Write-Log ("Action failed: {0}" -f $failure.Message) 'ERROR'
    [System.Windows.MessageBox]::Show($failure.Message,'Error','OK','Error') | Out-Null
    Set-Status ('Failed: ' + $failure.Message) 'Error'
    return
  }
  if ($null -ne $job.SuccessStatus) { Set-Status $job.SuccessStatus }
  elseif ($null -ne $job.Results) {
    $summary = Get-ResultSummary $job.Results
    if ($summary.Failed -gt 0) { Set-Status ('Done: {0}' -f $summary.Text) 'Error' }
    else { Set-Status ('Done: {0}' -f $summary.Text) }
  }
  else { Set-Status 'Done!' }

  # Runs after the UI is released, so it may open a dialog or start another action
  if ($job.OnSuccess) {
    try { & $job.OnSuccess $output }
    catch { [System.Windows.MessageBox]::Show($_.Exception.Message,'Error','OK','Error') | Out-Null }
  }
})

# Runs $Action in the worker runspace. $Action is re-parsed there, so it can only use
# its parameters and the functions in $WorkerFunctions, never UI controls.
# $OnSuccess runs on the UI thread with the action's output.
Function Start-CaseAction {
  param(
    [Parameter(Mandatory=$true)][string]$Action,
    [hashtable]$Arguments = @{},
    [System.Windows.Controls.DataGrid]$ResultsGrid,
    [AllowNull()][object]$SuccessStatus = $null,
    [scriptblock]$OnSuccess
  )
  $results = $null
  $summary = $null
  if ($ResultsGrid) {
    $results = $ResultsByGrid[$ResultsGrid.Name]; $results.Clear()
    $summary = $SummaryByGrid[$ResultsGrid.Name]; if ($summary) { $summary.Text = '' }
  }
  $Shared.Progress = $null
  $ps = [powershell]::Create()
  $ps.Runspace = $Worker
  $null = $ps.AddScript($Action.ToString(), $true).AddParameters($Arguments)   # local scope: no leftovers between actions

  $handle = $ps.BeginInvoke()
  Set-UIBusy $true; Show-Busy; Set-Status ''
  $script:Job = @{ PS = $ps; Handle = $handle; Results = $results; Summary = $summary; SuccessStatus = $SuccessStatus; OnSuccess = $OnSuccess }
  $PollTimer.Start()
}

# 'a', 'b', 'c' and 2 more
Function Join-Preview {
  param([string[]]$Items, [int]$Max = 3)
  $text = (@($Items | Select-Object -First $Max | ForEach-Object { "'$_'" })) -join ', '
  if ($Items.Count -gt $Max) { $text += ' and {0} more' -f ($Items.Count - $Max) }
  $text
}

Function Set-Hint {
  param([System.Windows.Controls.TextBox]$Box, [System.Windows.Controls.TextBlock]$Hint, [string]$Text, [ValidateSet('Normal','Warning','Invalid')`
][string]$Kind = 'Normal')
  $Hint.Text = $Text
  $key = switch ($Kind) { 'Warning' { 'WarningText' } 'Invalid' { 'DangerText' } default { 'TextSecondary' } }
  $Hint.SetResourceReference([System.Windows.Controls.TextBlock]::ForegroundProperty, $key)
  $Box.Tag = if ($Kind -eq 'Normal') { $null } else { $Kind }
}


Function Update-CaseHint {
  param([System.Windows.Controls.TextBox]$Box, [System.Windows.Controls.TextBlock]$Hint)
  $cases = @(ConvertTo-CaseList -CasesMultiline $Box.Text)
  if ($cases.Count -eq 0) { Set-Hint -Box $Box -Hint $Hint -Text ''; return }
  $count = if ($cases.Count -eq 1) { '1 case' } else { '{0} cases' -f $cases.Count }
  if (-not $script:CasesLoaded) {
    Set-Hint -Box $Box -Hint $Hint -Text ('{0}. Connect to check the names against the case list.' -f $count)
    return
  }
  $unknown = @($cases | Where-Object { -not $script:CaseNameSet.Contains($_) })
  if ($unknown.Count -gt 0) { Set-Hint -Box $Box -Hint $Hint -Text ('{0}. Not in the case list: {1}' -f $count, (Join-Preview $unknown)) -Kind `
'Warning' }
  else { Set-Hint -Box $Box -Hint $Hint -Text ('{0}, all found in the case list.' -f $count) }
}

Function Update-EmailHint {
  param([System.Windows.Controls.TextBox]$Box, [System.Windows.Controls.TextBlock]$Hint, [string]$EmptyText = '', [string]$InvalidNote = `
'these will be skipped')
  $entries = @(Split-InputList $Box.Text)
  if ($entries.Count -eq 0) { Set-Hint -Box $Box -Hint $Hint -Text $EmptyText; return }
  $valid = @(ConvertTo-EmailList -EmailsMultiline $Box.Text)
  $invalid = @($entries | Where-Object { $_ -notmatch $EmailPattern } | Select-Object -Unique)
  $count = if ($valid.Count -eq 1) { '1 reviewer' } else { '{0} reviewers' -f $valid.Count }
  if ($invalid.Count -gt 0) { Set-Hint -Box $Box -Hint $Hint -Text ('{0}. Not valid email addresses, {1}: {2}' -f $count, $InvalidNote, (`
Join-Preview $invalid)) -Kind 'Invalid' }
  else { Set-Hint -Box $Box -Hint $Hint -Text $count }
}

  } else {

    $HintRem_Replacement.Visibility = 'Collapsed'
  }
}

$script:AfterCaseLoad = $null

$LoadCasesAction = {
  Connect-Compliance
  $seen = @{}
  $loaded = 0
  foreach ($type in 'eDiscovery', 'AdvancedEdiscovery') {
    try { $cases = @(Get-ComplianceCase -CaseType $type -ErrorAction Stop) }
    catch { Write-Log ("Could not list {0} cases: {1}" -f $type, $_.Exception.Message) 'WARN'; continue }
    $loaded++
    foreach ($c in $cases) {
      $id = [string]$c.Identity
      if ($seen.ContainsKey($id)) { continue }
      $seen[$id] = $true
      $created = ''
      if ($c.CreatedDateTime) { $created = ([datetime]$c.CreatedDateTime).ToString('yyyy-MM-dd') }
      [pscustomobject]@{
        Name     = [string]$c.Name
        Type     = $(if ($type -eq 'AdvancedEdiscovery') { 'Premium' } else { 'Standard' })
        Status   = [string]$c.Status
        Created  = $created
        Identity = $id
      }
    }
  }
  if ($loaded -eq 0) { throw 'Could not list cases. Check the log for details.' }
}

Function Select-PowerToolsCase {
  param([string]$Search, [bool]$IncludeClosed)
  $Search = "$Search".Trim()
  @($script:CaseCache | Where-Object {
    ($IncludeClosed -or $_.Status -notlike 'Closed*') -and
    (-not $Search -or $_.Name.IndexOf($Search, [StringComparison]::OrdinalIgnoreCase) -ge 0)
  } | Sort-Object Name)
}

# Loads every case into the cache; $Then runs on the UI thread afterwards
Function Start-CaseLoad {
  param([scriptblock]$Then)
  $script:AfterCaseLoad = $Then
  Start-CaseAction -Action $LoadCasesAction -SuccessStatus '' -OnSuccess {
    param($out)
    $script:CaseCache = @($out)
    $script:CasesLoaded = $true
    $script:CaseNameSet.Clear()
    foreach ($c in $script:CaseCache) {
      if ($c.Name) { $null = $script:CaseNameSet.Add($c.Name) }
      if ($c.Identity) { $null = $script:CaseNameSet.Add($c.Identity) }
    }
    Update-CasesGrid
    Update-AllCaseHint
    Set-Status ('Loaded {0} cases' -f $script:CaseCache.Count)
    if ($script:AfterCaseLoad) { $next = $script:AfterCaseLoad; $script:AfterCaseLoad = $null; & $next }
  }
}

# Switches off one tool that failed to start, so the rest of PowerTools keeps working.
# The reason goes to the log and is shown once when the window opens.
$script:ModuleStartupFailures = New-Object 'System.Collections.Generic.List[string]'
Function Disable-PowerToolsModule {
  param([string]$Name, [object[]]$Controls, $ErrorRecord)
  $message = [string]$ErrorRecord.Exception.Message
  try {
    Write-Log ('Tool "{0}" failed to start and has been switched off: {1} | Line {2} | {3}' -f $Name, $message, `
$ErrorRecord.InvocationInfo.ScriptLineNumber, $ErrorRecord.ScriptStackTrace) 'ERROR'
  } catch { Write-PowerToolsCleanupWarning -Message ('Could not log tool start-up failure: ' + $_.Exception.Message) }
  foreach ($control in $Controls) {
    if ($control) {
      try { $control.IsEnabled = $false; $control.ToolTip = ('{0} is unavailable: {1}' -f $Name, $message) }
      catch { Write-PowerToolsCleanupWarning -Message ('Could not switch off a control: ' + $_.Exception.Message) }
    }
  }
  $script:ModuleStartupFailures.Add(('{0}: {1}' -f $Name, $message))
}

Function Get-OperatorErrorDetail {
  param([string]$Message,[int]$MaxLength=320)
  $text=([string]$Message).Trim() -replace '[\r\n\t]+',' ' -replace '\s{2,}',' '
  if([string]::IsNullOrWhiteSpace($text)){return 'Unknown error'}
  if($text.Length-gt$MaxLength){return ($text.Substring(0,$MaxLength).TrimEnd()+'...')}
  return $text
}

# Same saved-file status command used by the existing verification flow.
# Diagnostic wrapper: one call, no retries, no local-only fallback, no changed outcome.
Function Get-PowerToolsSavedFileLabelStatus {
  param([string]$Path,[string]$Context)
  $clock=[Diagnostics.Stopwatch]::StartNew()
  try {
    $command=Get-Command Get-FileStatus -ErrorAction Stop
    $file=Get-Item -LiteralPath $Path -ErrorAction Stop
    Write-Log ('Label verification START | Context={0}; SelfTest={1}; Path={2}; Bytes={3}; Module={4}; Version={5}; ExcelRunning={6}' -f `
$Context,[bool]$script:SelfTestActive,$file.FullName,$file.Length,$command.ModuleName,$command.Version,[bool](Get-Process EXCEL -ErrorAction `
SilentlyContinue)) 'INFO'
    $status=Get-FileStatus -Path $Path -ErrorAction Stop
    $clock.Stop()
    Write-Log ('Label verification RESPONSE | Context={0}; ElapsedMs={1}; IsLabeled={2}; SubLabelId={3}; RMSProtected={4}' -f $Context,`
$clock.ElapsedMilliseconds,$status.IsLabeled,$status.SubLabelId,$status.IsRMSProtected) 'INFO'
    return $status
  } catch {
    $clock.Stop()
    Write-Log ('Label verification ERROR | Context={0}; ElapsedMs={1}; Path={2}; ExceptionType={3}; Error={4}; Stack={5}' -f $Context,`
$clock.ElapsedMilliseconds,$Path,$_.Exception.GetType().FullName,$_.Exception.Message,$_.ScriptStackTrace) 'ERROR'
    throw
  }
}

Function Get-SelfTestExpectation {
  param([string]$Test)
  $expectationMatches=@($script:SelfTestExpectations.Tests | Where-Object Test -CEQ $Test)
  if($expectationMatches.Count -gt 1){throw ('Duplicate expected-results entries: '+$Test)}
  if($expectationMatches.Count -eq 1){return $expectationMatches[0]}
  return $null
}

Function Assert-SelfTestValue {
  param([string]$Name,$Expected,$Actual)
  if([string]$Expected -cne [string]$Actual){throw ('{0}: expected [{1}], actual [{2}]' -f $Name,$Expected,$Actual)}
}

Function Test-SelfTestWorkbookContent {
  param($Workbook)
  if(-not $script:SelfTestActive){return}
  $expectation=Get-SelfTestExpectation $script:SelfTestCaseName
  if(-not $expectation -or -not $expectation.Worksheets){return}
  Write-PowerToolsResourceSnapshot -Stage 'Workbook assertion inspection START'
  foreach($spec in $expectation.Worksheets){
    $sheet=$null;$used=$null;$range=$null
    try{
      if(-not $spec.Name){throw 'Worksheet expectation requires Name.'}
      $sheet=$Workbook.Worksheets.Item([string]$spec.Name)
      if($null -ne $spec.DataRowCount){
        if($null -eq $spec.DataStartRow -or [int]$spec.DataStartRow -lt 1){throw 'DataRowCount requires a positive DataStartRow.'}
        $used=$sheet.UsedRange
        $count=[Math]::Max(0,([int]$used.Row+[int]$used.Rows.Count-[int]$spec.DataStartRow))
        Assert-SelfTestValue ($spec.Name+' data rows') $spec.DataRowCount $count
      }
      foreach($cell in $spec.Cells){
        if(-not $cell.Address -or -not $cell.PSObject.Properties['Value']){throw 'Cell expectation requires Address and Value.'}
        $range=$sheet.Range([string]$cell.Address)
        if($range.Cells.Count -ne 1){throw 'Cell Address must identify exactly one cell.'}
        Assert-SelfTestValue ($spec.Name+'!'+$cell.Address) $cell.Value $range.Value2
        $range=$null
      }
      Add-SelfTestResult ($script:SelfTestCaseName+' / '+$spec.Name+' content') 'Expected content' 'Success' (`
'Worksheet, configured row count and cell values match the reviewed expectations. Checked in the open ' + 'workbook; saved file is not reopened.')
    }catch{
      $script:SelfTestContentFailed=$true
      Add-SelfTestResult ($script:SelfTestCaseName+' / '+$spec.Name+' content') 'Expected content' 'Failed' $_.Exception.Message
    }finally{
      # Borrowed workbook objects remain owned by the conversion controller.
      $range=$null;$used=$null;$sheet=$null
    }
  }
  $script:SelfTestWorkbookChecked=$true
  Write-PowerToolsResourceSnapshot -Stage 'Workbook assertion inspection END'
}

    $replacement = $TxtRem_Replacement.Text.Trim()
    if ($replacement -and $replacement -notmatch $EmailPattern) { throw 'The replacement reviewer is not a valid email address.' }
    # A replacement is optional for targeted removal. It remains mandatory when the
    # reviewers-to-remove box is blank because blank means remove all current members.
    if ($emails.Count -eq 0 -and -not $replacement) { throw (`
'Please enter a replacement reviewer email when removing all reviewers. For targeted removal, enter one or ' + `
'more reviewers to remove and the replacement may be left blank.') }
  } catch { Show-InputError $_.Exception.Message; return }

  Start-CaseAction -ResultsGrid $GridRem_Results -Arguments @{ Cases = $cases; Emails = [string[]]$emails; Replacement = $replacement } -Action {
    param([string[]]$Cases, [string[]]$Emails, [string]$Replacement)
    Connect-Compliance

    $i = 0
    foreach ($case in $Cases) {
      Set-ActionProgress -Done $i -Total $Cases.Count; $i++

      # Ensure replacement is present first when one was supplied.
      if ($Replacement) {
        try {
          Add-ComplianceCaseMember -Case $case -Member $Replacement -ErrorAction SilentlyContinue
          Write-Result -Case $case -Member $Replacement -Action 'Add replacement' -Result Info -Detail 'Added, or already a member'
        }
        catch {
          Write-Result -Case $case -Member $Replacement -Action 'Add replacement' -Result Failed -Detail $_.Exception.Message
        }
      }

      # Determine targets to remove
      $targets = $null
      if ($Emails -and $Emails.Count -gt 0) {
        # Targeted removal (exclude replacement if included)
        $targets = $Emails | Where-Object { $_ -and (-not $Replacement -or ($_ -ine $Replacement)) }
      } else {
        # Remove ALL current members except the replacement
        try {
          $targets = @(Get-CaseMemberEmail -Case $case -ExcludeEmail $Replacement)
          if ($targets.Count -eq 0) {
            Write-Result -Case $case -Action 'Remove reviewers' -Result Info -Detail `
'No removable members found (after excluding replacement)'
            continue
          } else {
            Write-Result -Case $case -Action 'Remove reviewers' -Result Info -Detail (`
'Removing all {0} current members except the replacement' -f $targets.Count)
          }
        }
        catch {
          Write-Result -Case $case -Action 'Remove reviewers' -Result Failed -Detail ("Failed to enumerate members: {0}" -f $_)
          continue
        }
      }

      foreach ($m in $targets) {
        if ($Replacement -and $m -ieq $Replacement) {
          Write-Result -Case $case -Member $m -Action 'Remove reviewer' -Result Skipped -Detail 'Replacement reviewer is kept'
          continue
        }
        try {
          Remove-ComplianceCaseMember -Case $case -Member $m -Confirm:$false -ErrorAction Stop
          Write-Result -Case $case -Member $m -Action 'Remove reviewer' -Result Success
        }
        catch {
          Write-Result -Case $case -Member $m -Action 'Remove reviewer' -Result Failed -Detail $_.Exception.Message
        }
      }
    }
    Set-ActionProgress -Done $i -Total $Cases.Count
  }
})

Function Invoke-TeamsConversion {
  param([string]$InputPath,[string]$OutputFolder,[bool]$ApplyTeamsHC)
  $conversionStage = 'Validating input'
  $inputPath=$InputPath.Trim();$outputFolder=$OutputFolder.Trim()
  if (-not (Test-Path -LiteralPath $inputPath -PathType Container)) { Show-InputError 'Select a valid input folder containing .msg files.'; return }
  if (-not $outputFolder) { Show-InputError 'Select an output folder.'; return }
  if (-not (Test-Path -LiteralPath $outputFolder -PathType Container)) { try { New-Item -ItemType Directory -Path $outputFolder -Force `
-ErrorAction Stop | Out-Null } catch { Show-InputError ('Could not create output folder: {0}' -f $_.Exception.Message); return } }
  $stamp = Get-Date -Format 'yyyyMMdd_HHmmss'
  $outputXlsx = Join-Path $outputFolder ('TeamsChats_{0}.xlsx' -f $stamp)
  $outputCsv = Join-Path $outputFolder ('TeamsChats_{0}.csv' -f $stamp)
  try { $teamFiles = @(Get-ChildItem -LiteralPath $inputPath -Filter '*.msg' -File -ErrorAction Stop) }
  catch { Show-InputError ('Could not enumerate MSG files: {0}' -f $_.Exception.Message); return }
  $msgCount = $teamFiles.Count
  if ($msgCount -eq 0) { Show-InputError 'No .msg files were found in the selected input folder.'; return }
  if (Get-Process OUTLOOK -ErrorAction SilentlyContinue) { Update-TeamsOutlookStatus; return }
  try {
    $script:TeamsConversionRunning = $true
    $TeamsOutlookTimer.Stop()
    $BorderTeams_Outlook.Visibility = 'Collapsed'
    $TeamsResults.Clear(); $SumTeams.Text = 'Waiting for Outlook profile / starting conversion...'
    Set-UIBusy $true; Show-Busy; Set-Status 'Waiting for Outlook profile / starting conversion...' 'Neutral'
    $outlook = $null; $session = $null; $item = $null
    $testOutlookOwner=$null;$testOutlookQuitRequested=$false
    try {
      $conversionStage = 'Starting Outlook / waiting for profile if required'
      Write-Log ('Teams to Excel stage: {0}' -f $conversionStage)
      $SumTeams.Text = 'Starting Outlook. If an Outlook profile prompt appears, select the required profile to continue...'
      Set-Status 'Waiting for Outlook profile selection if required...' 'Neutral'
      # Recheck immediately before COM activation; never adopt a pre-existing session.
      $testActivationTime=Get-Date
      if($script:SelfTestActive -and (Get-Process OUTLOOK -ErrorAction SilentlyContinue)){
        throw 'Outlook appeared before self-test activation. Session left untouched.'
      }
      $outlook = New-Object -ComObject Outlook.Application
      if($script:SelfTestActive){
        $candidates=@(Get-Process OUTLOOK -ErrorAction SilentlyContinue)
        if($candidates.Count -eq 1 -and $candidates[0].StartTime -ge $testActivationTime){
          $testOutlookOwner=[pscustomobject]@{Id=$candidates[0].Id;StartTime=$candidates[0].StartTime}
          Write-Log ('Self-test Outlook ownership recorded: PID={0}; StartTime={1:o}' -f $testOutlookOwner.Id,$testOutlookOwner.StartTime)
        }else{Write-Log 'Self-test Outlook ownership could not be established; automatic shutdown disabled.' 'WARN'}
      }

      $conversionStage = 'Awaiting Outlook profile selection'
      Write-Log ('Teams to Excel stage: {0}' -f $conversionStage)
      $SumTeams.Text = 'Awaiting Outlook profile selection...'
      Set-Status 'Awaiting Outlook profile selection...' 'Neutral'; Update-PowerToolsUI
      $teamDone = 0
      $progressClock = [Diagnostics.Stopwatch]::StartNew()
      $progressIntervalMs = 250
      Set-ConversionProgress -Done 0 -Total $teamFiles.Count -Unit 'messages'
      $results = foreach ($file in $teamFiles) {
        $item = $null
        try {
          if ($null -eq $session) { $session = $outlook.Session }
          $item = $session.OpenSharedItem($file.FullName)
          if ($teamDone -eq 0) {
            $conversionStage = 'Reading MSG files'
            Write-Log ('Teams to Excel stage: {0}' -f $conversionStage)
          }
          $senderEmail = $item.SenderEmailAddress
          $to = $item.To
          $sentOn = $item.SentOn
          [pscustomobject]@{ SenderEmail=$senderEmail; To=$to; SentOn=$(if($sentOn){$sentOn}else{$null}); Body=$item.Body; __File=$file.Name; __Result='Success'; __Detail='Converted to Excel' }
        } catch {
          $operatorError=Get-OperatorErrorDetail $_.Exception.Message
          Write-Log ('Teams MSG to Excel - File={0}; Stage={1}; Error={2}; ExceptionType={3}; Stack={4}' -f $file.FullName,`
$conversionStage,$_.Exception.Message,$_.Exception.GetType().FullName,$_.ScriptStackTrace) 'ERROR'
          [pscustomobject]@{ SenderEmail=$null; To=$null; SentOn=$null; Body=('ERROR processing {0}: {1}' -f $file.Name,$operatorError); `
__File=$file.Name; __Result='Failed'; __Detail=('MSG conversion failed - {0}' -f $operatorError) }
        } finally {
          if ($item) { [void][Runtime.InteropServices.Marshal]::ReleaseComObject($item); $item=$null }
          $teamDone++
          if ($teamDone -eq $msgCount -or $progressClock.ElapsedMilliseconds -ge $progressIntervalMs) {
            $statusText = 'Processed {0} of {1} messages. Last: {2}' -f $teamDone, $msgCount, $file.Name
            $SumTeams.Text = $statusText
            Set-Status $statusText 'Neutral'
            # This helper already pumps the WPF dispatcher.
            Set-ConversionProgress -Done $teamDone -Total $msgCount -Unit 'messages'
            $progressClock.Restart()
          }
        }
      }
      $conversionStage = 'MSG extraction complete'
      Write-Log ('Teams conversion stage: {0}' -f $conversionStage)
      if (-not $applyTeamsHC) {
        $conversionStage = 'Preparing unlabelled CSV output'
        $SumTeams.Text = 'MSG extraction complete. Preparing unlabelled CSV output...'
        Set-Status 'MSG extraction complete. Preparing unlabelled CSV output...' 'Neutral'; Update-PowerToolsUI
        Write-Log ('Teams conversion stage: {0}' -f $conversionStage)
        $csvRows = @($results | Select-Object SenderEmail,To,SentOn,Body)
        $conversionStage = 'Writing CSV'
        $SumTeams.Text = ('Writing {0} messages to unlabelled CSV...' -f $csvRows.Count)
        Set-Status ('Writing {0} messages to unlabelled CSV...' -f $csvRows.Count) 'Neutral'; Update-PowerToolsUI
        $csvRows | Export-Csv -LiteralPath $outputCsv -NoTypeInformation -Encoding UTF8 -Force
        if (-not (Test-Path -LiteralPath $outputCsv -PathType Leaf)) { throw ('CSV output was not created: {0}' -f $outputCsv) }
        $csvFile = Get-Item -LiteralPath $outputCsv -ErrorAction Stop
        Write-Log ('Unlabelled CSV output created by user selection: {0} | {1:N0} bytes' -f $csvFile.FullName,$csvFile.Length) 'WARN'
        $outputXlsx = $csvFile.FullName
        $SumTeams.Text = ('CSV saved without sensitivity label: {0}' -f $csvFile.FullName)
        Set-Status 'Teams to CSV conversion complete. No sensitivity label applied.' 'Success'; Update-PowerToolsUI
      } else {
        $SumTeams.Text = 'MSG extraction complete. Preparing Excel workbook...'
        Set-Status 'MSG extraction complete. Preparing Excel workbook...' 'Neutral'; Update-PowerToolsUI
        $excel = $null; $workbook = $null; $worksheet = $null
        try {
        $conversionStage = 'Starting Excel'
        Write-Log ('Teams to Excel stage: {0}' -f $conversionStage)
        $SumTeams.Text = 'Starting Excel...'
        Set-Status 'Starting Excel...' 'Neutral'; Update-PowerToolsUI
        $excel = New-Object -ComObject Excel.Application
        $excel.Visible = $false
        $excel.DisplayAlerts = $false
        $conversionStage = 'Creating workbook'
        Write-Log ('Teams to Excel stage: {0}' -f $conversionStage)
        $SumTeams.Text = 'Creating Excel workbook...'
        Set-Status 'Creating Excel workbook...' 'Neutral'; Update-PowerToolsUI
        $workbook = $excel.Workbooks.Add()
        $worksheet = $workbook.Worksheets.Item(1)
        $worksheet.Name = 'Teams Chats'
        $conversionStage = 'Writing workbook'
        Write-Log ('Teams to Excel stage: {0}' -f $conversionStage)
        $SumTeams.Text = ('Writing {0} messages to Excel workbook...' -f $results.Count)
        Set-Status ('Writing {0} messages to Excel workbook...' -f $results.Count) 'Neutral'; Update-PowerToolsUI
        $headers = @('SenderEmail','To','SentOn','Body')
        for ($c = 0; $c -lt $headers.Count; $c++) { $worksheet.Cells.Item(1,$c+1) = $headers[$c] }
        $row = 2
        foreach ($r in $results) {
          $worksheet.Cells.Item($row,1) = [string]$r.SenderEmail
          $worksheet.Cells.Item($row,2) = [string]$r.To
          if ($r.SentOn) { $worksheet.Cells.Item($row,3) = $r.SentOn.ToString('yyyy-MM-dd HH:mm:ss') }
          $worksheet.Cells.Item($row,4) = [string]$r.Body
          $row++
        }
        [void]$worksheet.Columns.Item(1).AutoFit()
        [void]$worksheet.Columns.Item(2).AutoFit()
        [void]$worksheet.Columns.Item(3).AutoFit()
        $worksheet.Columns.Item(4).ColumnWidth = 80
        $worksheet.Columns.Item(4).WrapText = $true
        if ($applyTeamsHC) {
          $conversionStage = 'Applying Highly Confidential label in Excel'
          Write-Log ('Teams to Excel stage: {0}' -f $conversionStage)
          $SumTeams.Text = ('Applying Highly Confidential | {0} label in Excel...' -f $HighlyConfidentialLabelName)
          Set-Status ('Applying Highly Confidential | {0} label in Excel...' -f $HighlyConfidentialLabelName) 'Neutral'; Update-PowerToolsUI
          try {
            $sensitivityLabel = $workbook.SensitivityLabel
            if (-not $sensitivityLabel) { throw 'Excel did not expose the Workbook.SensitivityLabel API.' }
            $labelInfo = $sensitivityLabel.CreateLabelInfo()
            $labelInfo.AssignmentMethod = 1
            $labelInfo.LabelId = [string]$HighlyConfidentialLabelId
            $labelInfo.LabelName = $HighlyConfidentialLabelName
            $labelInfo.IsEnabled = $true
            $labelInfo.SetDate = Get-Date
            $sensitivityLabel.SetLabel($labelInfo, $labelInfo)
            Write-Log ('Excel native sensitivity label assigned: LabelId={0}; LabelName={1}; AssignmentMethod=Privileged' -f `
$HighlyConfidentialLabelId,$HighlyConfidentialLabelName)
          } catch {
            throw ('Excel native SensitivityLabel API failed: {0}' -f $_.Exception.Message)
          } finally {
            if ($labelInfo) { try { [Runtime.InteropServices.Marshal]::ReleaseComObject($labelInfo) | Out-Null } catch { `
Write-PowerToolsCleanupWarning -Message ('Best-effort operation failed at line {0}: {1}' -f $_.InvocationInfo.ScriptLineNumber, $_.Exception.Message)`
 }; $labelInfo=$null }
            if ($sensitivityLabel) { try { [Runtime.InteropServices.Marshal]::ReleaseComObject($sensitivityLabel) | Out-Null } catch { `
Write-PowerToolsCleanupWarning -Message ('Best-effort operation failed at line {0}: {1}' -f $_.InvocationInfo.ScriptLineNumber, $_.Exception.Message)`
 }; $sensitivityLabel=$null }
          }
        } else {
          $conversionStage = 'Sensitivity labelling skipped'
          Write-Log 'Teams to Excel: Highly Confidential labelling was not selected.'
          $SumTeams.Text = 'Sensitivity labelling not selected. Preparing unlabelled Excel workbook...'
          Set-Status 'Sensitivity labelling not selected. Preparing unlabelled Excel workbook...' 'Neutral'; Update-PowerToolsUI
        }

        $conversionStage = $(if($applyTeamsHC){'Saving labelled XLSX'}else{'Saving XLSX without sensitivity label'})
        Write-Log ('Teams to Excel stage: {0} -> {1}' -f $conversionStage,$outputXlsx)
        $SumTeams.Text = ('Saving {0}Excel workbook: {1}' -f $(if($applyTeamsHC){'labelled '}else{''}),[IO.Path]::GetFileName($outputXlsx))
        Set-Status ('Saving {0}Excel workbook: {1}' -f $(if($applyTeamsHC){'labelled '}else{''}),[IO.Path]::GetFileName($outputXlsx)) `
'Neutral'; Update-PowerToolsUI
        $requestedXlsxPath = [System.IO.Path]::GetFullPath($outputXlsx)
        $workbook.SaveAs($requestedXlsxPath, 51)
        $actualXlsxPath = [string]$workbook.FullName
        if ([string]::IsNullOrWhiteSpace($actualXlsxPath)) { $actualXlsxPath = $requestedXlsxPath }
        if (-not [System.IO.Path]::IsPathRooted($actualXlsxPath)) {
          $actualXlsxPath = Join-Path $outputFolder $actualXlsxPath
        }
        $actualXlsxPath = [System.IO.Path]::GetFullPath($actualXlsxPath)
        Write-Log ('Excel SaveAs requested path: {0}' -f $requestedXlsxPath)
        Write-Log ('Excel Workbook.FullName after SaveAs: {0}' -f $actualXlsxPath)
        Test-SelfTestWorkbookContent -Workbook $workbook
        $workbook.Close($false)
        [Runtime.InteropServices.Marshal]::ReleaseComObject($workbook) | Out-Null; $workbook = $null
        $excel.Quit()
        [Runtime.InteropServices.Marshal]::ReleaseComObject($worksheet) | Out-Null; $worksheet = $null
        [Runtime.InteropServices.Marshal]::ReleaseComObject($excel) | Out-Null; $excel = $null
        [GC]::Collect(); [GC]::WaitForPendingFinalizers()
        } finally {
          if ($worksheet) { try { [Runtime.InteropServices.Marshal]::ReleaseComObject($worksheet) | Out-Null } catch { `
Write-PowerToolsCleanupWarning -Message ('Best-effort operation failed at line {0}: {1}' -f $_.InvocationInfo.ScriptLineNumber, $_.Exception.Message)`
 }; $worksheet=$null }
          if ($workbook) { try { $workbook.Close($false) } catch { Write-PowerToolsCleanupWarning -Message (`
'Best-effort operation failed at line {0}: {1}' -f $_.InvocationInfo.ScriptLineNumber, $_.Exception.Message) }; try { `
[Runtime.InteropServices.Marshal]::ReleaseComObject($workbook) | Out-Null } catch { Write-PowerToolsCleanupWarning -Message (`
'Best-effort operation failed at line {0}: {1}' -f $_.InvocationInfo.ScriptLineNumber, $_.Exception.Message) }; $workbook=$null }
          if ($excel) { try { $excel.Quit() } catch { Write-PowerToolsCleanupWarning -Message ('Best-effort operation failed at line {0}: {1}' `
-f $_.InvocationInfo.ScriptLineNumber, $_.Exception.Message) }; try { [Runtime.InteropServices.Marshal]::ReleaseComObject($excel) | Out-Null } catch `
{ Write-PowerToolsCleanupWarning -Message ('Best-effort operation failed at line {0}: {1}' -f $_.InvocationInfo.ScriptLineNumber, `
$_.Exception.Message) }; $excel=$null }
          [GC]::Collect(); [GC]::WaitForPendingFinalizers()
        }
        $conversionStage = 'Verifying Excel saved path'
        if ([string]::IsNullOrWhiteSpace($actualXlsxPath)) { $actualXlsxPath = [System.IO.Path]::GetFullPath($outputXlsx) }
        Write-Log ('Teams to Excel stage: {0} -> {1}' -f $conversionStage,$actualXlsxPath)
        $SumTeams.Text = ('Verifying Excel saved path: {0}' -f $actualXlsxPath)
        Set-Status 'Verifying Excel actual saved path...' 'Neutral'; Update-PowerToolsUI
        if (-not [System.IO.File]::Exists($actualXlsxPath)) {
          $requested = [System.IO.Path]::GetFullPath($outputXlsx)
          throw ('Excel SaveAs completed but workbook was not found. Requested=[{0}] Excel.FullName=[{1}]' -f $requested,$actualXlsxPath)
        }
        $outputXlsx = $actualXlsxPath
        $savedXlsx = Get-Item -LiteralPath $actualXlsxPath -ErrorAction Stop
        Write-Log ('Verified XLSX exists: {0} | {1:N0} bytes | LastWriteTime={2:o}' -f $savedXlsx.FullName,$savedXlsx.Length,`
$savedXlsx.LastWriteTime)

Function Invoke-PdfConversion {
  param([string]$InputPath,[string]$OutputFolder,[bool]$ApplyPdfHC,[string]$ImageWidth=$PdfDefaults.ImageWidth,`
[string]$ImageHeight=$PdfDefaults.ImageHeight,[string]$MaxWidth=$PdfDefaults.MaxWidth,[string]$Zoom=$PdfDefaults.Zoom,`
[string]$Quality=$PdfDefaults.Quality)
  $in=$InputPath.Trim();$out=$OutputFolder.Trim()
  $imgWidth=0; $imgHeight=0.0; $mw=0; $z=0; $q=0
  if(-not(Test-Path -LiteralPath $in -PathType Container)){Show-InputError 'Select a valid input folder containing .html files.';return}
  if(-not $out){Show-InputError 'Select an output folder.';return}
  if(-not[int]::TryParse($ImageWidth.Trim(),[ref]$imgWidth)-or$imgWidth-lt1-or$imgWidth-gt100){Show-InputError `
'Maximum displayed image width must be 1-100 percent.';return}
  if(-not[double]::TryParse($ImageHeight.Trim(),[System.Globalization.NumberStyles]::Float,[System.Globalization.CultureInfo]::CurrentCulture,`
[ref]$imgHeight)-and -not [double]::TryParse($ImageHeight.Trim(),[System.Globalization.NumberStyles]::Float,`
[System.Globalization.CultureInfo]::InvariantCulture,[ref]$imgHeight)){Show-InputError 'Maximum displayed image height must be a number.';return}
  if($imgHeight-lt1-or$imgHeight-gt100){Show-InputError 'Maximum displayed image height must be 1-100 cm.';return}
  if(-not[int]::TryParse($MaxWidth.Trim(),[ref]$mw)-or$mw-lt100-or$mw-gt10000){Show-InputError 'Maximum image resolution must be 100-10000 pixels.'`
;return}
  if(-not[int]::TryParse($Zoom.Trim(),[ref]$z)-or$z-lt25-or$z-gt200){Show-InputError 'Zoom must be 25-200.';return}
  if(-not[int]::TryParse($Quality.Trim(),[ref]$q)-or$q-lt1-or$q-gt100){Show-InputError 'JPEG quality must be 1-100.';return}
  if(-not(Test-Path -LiteralPath $out)){New-Item -ItemType Directory -Path $out -Force|Out-Null}
  $files=@(Get-ChildItem -LiteralPath $in -Filter '*.html' -File|Where-Object Name -notlike '_tmp_*'); if(-not$files){Show-InputError `
'No .html files found.';return}
  Write-PdfConversionLog (('Starting conversion. Input={0}; Output={1}; Files={2}; HCLabel={3}; ImageWidth={4}%; ImageHeight={5}cm; ' + `
'MaxResolution={6}px; Zoom={7}%; JPEGQuality={8}')-f$in,$out,$files.Count,$applyPdfHC,$imgWidth,$imgHeight,$mw,$z,$q)
  $TeamsPdfResults.Clear(); $SumTeamsPdf.Text='Preparing {0} HTML file(s)...'-f$files.Count; Set-UIBusy $true; Show-Busy; Set-ConversionProgress `
-Done 0 -Total $files.Count -Unit 'files'; Set-Status ('Preparing {0} HTML file(s) for PDF conversion...'-f$files.Count) 'Neutral'; `
Update-PowerToolsUI; $pdfDone=0
  try{

|Where-Object{Test-Path -LiteralPath $_}|Select-Object -First 1; if(-not$edge){throw'Microsoft Edge (msedge.exe) was not found.'}
    Write-PdfConversionLog ('Microsoft Edge={0}'-f$edge)
    $pdfRunTemp=Join-Path $env:TEMP ('PDFPowerConvert_'+[guid]::NewGuid().ToString('N'));New-Item -ItemType Directory -Path $pdfRunTemp -Force `
-ErrorAction Stop|Out-Null;$edgeProfilePath=Join-Path $pdfRunTemp 'EdgeProfile';$log=Join-Path $pdfRunTemp 'Edge.log'; $ok=0; $bad=0
    $inject=@'
<style>
  html { zoom: __ZOOM__%; }
  img, canvas { max-width: __IMGW__% !important; max-height: __IMGH__cm !important; height: auto !important; object-fit: contain; }
</style>
<script nonce="__NONCE__">
(function(){var MAXW=__MAXW__,Q=__Q__/100,LIMIT=500*1024;function shrink(img){var w=img.naturalWidth,h=img.naturalHeight;if(!w||!h)return;var
bigData=img.src.indexOf('data:')===0&&img.src.length*0.75>LIMIT;if(w<=MAXW&&!bigData)return;var nw=Math.min(w,MAXW),nh=Math.round(h*nw/w),
c=document.createElement('canvas');c.width=nw;c.height=nh;var g=c.getContext('2d');g.fillStyle='#fff';g.fillRect(0,0,nw,nh);

c.className=img.className;img.parentNode.replaceChild(c,img)}}window.addEventListener('load',function(){var imgs=Array.prototype.slice.call(

</script>
'@

    $inject=$inject.Replace('__ZOOM__',"$z").Replace('__MAXW__',"$mw").Replace('__IMGW__',"$imgWidth").Replace('__IMGH__',$heightText).Replace(`
'__Q__',"$q")

# Local resource schemes remain allowed for export compatibility: this is
# a page CSP, not an OS sandbox or complete network isolation.
$nonceBytes = New-Object byte[] 16
$nonceGenerator = [Security.Cryptography.RandomNumberGenerator]::Create()

try {
  $nonceGenerator.GetBytes($nonceBytes)
}
finally {
  $nonceGenerator.Dispose()
}

$nonce = [System.Convert]::ToBase64String($nonceBytes)

       "style-src 'unsafe-inline' file: data:; img-src data: file: blob: https:; " +
       "font-src data: file:; media-src data: file:; connect-src 'none'; " +
       "object-src 'none'; frame-src 'none'; worker-src 'none'; " +
       "base-uri 'none'; form-action 'none'"

$inject = $inject.Replace('__NONCE__', $nonce)

  "<!DOCTYPE html>`r`n<meta charset=`"utf-8`">`r`n" +

  $inject

Write-PdfConversionLog (

  'toolbox optimiser nonce authorised; local resource schemes retained.'
)

          $edgeArguments = @(
            '--headless', '--disable-gpu', '--no-pdf-header-footer', '--log-level=3', '--virtual-time-budget=15000',
            ('--user-data-dir="{0}"' -f $edgeProfilePath),
            ('--print-to-pdf="{0}"' -f $pdf),
            ('"{0}"' -f $tmp)
          )
          $edgeProcess = Start-Process $edge -Wait -WindowStyle Hidden -PassThru -RedirectStandardError $log -ArgumentList $edgeArguments
          Write-PdfConversionLog ('Edge render completed: ExitCode={0}; DestinationExists={1}' -f $edgeProcess.ExitCode, (Test-Path -LiteralPath $pdf -PathType Leaf))

          if (Test-Path -LiteralPath $pdf) {

            $pdfName = [IO.Path]::GetFileName($pdf)
            Write-PdfConversionLog ('PDF created: {0}; Size={1:N0} KB' -f $pdf, $kb)

            if (-not $applyPdfHC) {
              # No label requested: the PDF is finished.
              $ok++
              Set-Status ('PDF created without sensitivity label, as selected: {0}' -f $pdfName) 'Neutral'

              # Apply the Highly Confidential label, then read it back from the saved file to prove it stuck.
              try {

                Update-PowerToolsUI

                  if (-not $pdfLabelStatus.IsLabeled -or [string]$pdfLabelStatus.SubLabelId -ne [string]$HighlyConfidentialLabelId) {

                  }
                }
                $ok++

              }
              catch {

            }
          }

              $f.FullName, $edgeExit, $edgeError) 'ERROR'
            $TeamsPdfResults.Add([pscustomobject]@{ File = $f.Name; Result = 'Failed'; Detail = $detail })
          }
        }
        catch {
          # Anything else that went wrong with this one file. The loop carries on with the next file.
          $bad++
          $operatorError = Get-OperatorErrorDetail $_.Exception.Message
          Write-PdfConversionLog ('File={0}; Destination={1}; Stage=Conversion; Error={2}; ExceptionType={3}; Stack={4}' -f `
            $f.FullName, $pdf, $_.Exception.Message, $_.Exception.GetType().FullName, $_.ScriptStackTrace) 'ERROR'
          $TeamsPdfResults.Add([pscustomobject]@{ File = $f.Name; Result = 'Failed'; Detail = ('PDF creation failed - ' + $operatorError) })
        }
        finally {
          # Always remove the temporary copy and move the progress bar on.
          if (Test-Path -LiteralPath $tmp) { Remove-Item -LiteralPath $tmp -Force -ErrorAction SilentlyContinue }
          $pdfDone++
          Set-ConversionProgress -Done $pdfDone -Total $files.Count -Unit 'files'
        }
      }

    finally {
      if (Test-Path -LiteralPath $header) { Remove-Item -LiteralPath $header -Force -ErrorAction SilentlyContinue }
    }

    Write-PdfConversionLog ('Completed. Files={0}; Succeeded={1}; Failed={2}; Output={3}' -f $files.Count, $ok, $bad, $out) $logLevel
    $SumTeamsPdf.Text = 'Processed {0} HTML file(s); {1} succeeded, {2} failed. Output: {3}' -f $files.Count, $ok, $bad, $out
    $statusKind = if ($bad) { 'Error' } else { 'Success' }

  }
  catch {

    Write-PdfConversionLog ('Overall conversion failure: Error={0}; ExceptionType={1}; Stack={2}' -f `
      $_.Exception.Message, $_.Exception.GetType().FullName, $_.ScriptStackTrace) 'ERROR'

    Set-Status 'Teams HTML to PDF failed' 'Error'
    Show-PowerToolsDialog @($_.Exception.Message, 'Teams HTML to PDF', 'OK', 'Error') | Out-Null
  }
  finally {

    if ($pdfRunTemp -and (Test-Path -LiteralPath $pdfRunTemp)) {

        Remove-Item -LiteralPath $pdfRunTemp -Recurse -Force -ErrorAction Stop
        Write-PdfConversionLog ('Removed per-run temporary workspace: ' + $pdfRunTemp)
      }
      catch {
        Write-PdfConversionLog ('Could not remove per-run temporary workspace {0}: {1}' -f $pdfRunTemp, $_.Exception.Message) 'WARN'
      }
    }
    Hide-Busy
    Set-UIBusy $false
    Update-TeamsOutlookStatus
  }
}

 Function Add-SelfTestResult {
  param([string]$Test,[string]$Area,[ValidateSet('Success','Failed','Skipped','Info')][string]$Result,[string]$Detail)
  $SelfTestResults.Add([pscustomobject]@{ Test=$Test; Area=$Area; Result=$Result; Detail=$Detail })

# Expected results: Resources\SelfTestExpectedResults.json (manually reviewed baseline).
# Schema: {"SchemaVersion":1,"Tests":[{"Test":"exact displayed case name",
# "OutputCount":1,"Worksheets":[{"Name":"sheet name","DataStartRow":2,
# "DataRowCount":0,"Cells":[{"Address":"A1","Value":"expected text"}]}],
# "Csv":{"RowCount":0,"Values":[{"Row":1,"Column":"column name","Value":"expected text"}]}}]}
# RowCount values above illustrate the schema only, not fixture expectations.
# Worksheet DataRowCount uses UsedRange's final row minus DataStartRow + 1.
# Cells use Excel Value2; Excel dates must be supplied as their serial values.

Function Test-SelfTestOutputContent {
  param([string]$Test,[object[]]$Files)
  $spec=Get-SelfTestExpectation $Test
  # Omit unconfigured PDF baseline rows; keep conversion/count checks and capture mode.
  if($Test -match '^Pdf HC=(True|False)$' -and ($null -eq $spec -or ($null -eq $spec.OutputCount -and -not $spec.Worksheets -and -not $spec.Csv))){`
return}
  if(-not $spec){Add-SelfTestResult ($Test+' content') 'Expected content' 'Skipped' `
'No reviewed expectations configured in Resources\SelfTestExpectedResults.json.';return}
  $checks=0
  if($null -ne $spec.OutputCount){Assert-SelfTestValue 'Output file count' $spec.OutputCount $Files.Count;$checks++}
  if($spec.Worksheets){
    if(-not $script:SelfTestWorkbookChecked){throw 'Worksheet expectations configured but no workbook content check was completed.'}
    if($script:SelfTestContentFailed){throw 'One or more workbook content assertions failed; see detailed results.'}
    $checks++
  }
  if($spec.Csv){
    $csvFiles=@($Files|Where-Object Extension -eq '.csv')
    if($csvFiles.Count -ne 1){throw 'CSV assertions require exactly one CSV output.'}
    $rows=@(Import-Csv -LiteralPath $csvFiles[0].FullName -ErrorAction Stop)
    if($null -ne $spec.Csv.RowCount){Assert-SelfTestValue 'CSV row count' $spec.Csv.RowCount $rows.Count;$checks++}
    foreach($value in $spec.Csv.Values){
      if([int]$value.Row -lt 1 -or [int]$value.Row -gt $rows.Count){throw 'CSV expected row is outside the output.'}
      $property=$rows[[int]$value.Row-1].PSObject.Properties[[string]$value.Column]
      if(-not $property -or -not $value.PSObject.Properties['Value']){throw 'CSV expectation requires an existing Column and Value.'}
      Assert-SelfTestValue ('CSV row '+$value.Row+' / '+$value.Column) $value.Value $property.Value;$checks++
    }
  }
  if($checks -eq 0){Add-SelfTestResult ($Test+' content') 'Expected content' 'Skipped' 'Expectation entry contains no supported assertions.';return}
  Add-SelfTestResult ($Test+' expected results') 'Expected content' 'Success' (`
'{0} configured assertion group(s) passed. PDF text extraction is not implemented.' -f $checks)
}

# ------------------ Keyboard shortcuts ------------------
# Ctrl+Enter runs the current page, F5 refreshes the case list, Ctrl+1 to Ctrl+9 switch tools, Ctrl+0 opens Settings
$window.Add_PreviewKeyDown({
  param($s, $e)
  $null = $s # Preserve the sender argument slot; only event arguments are used.
  # Ctrl without Alt, so AltGr (Ctrl+Alt) still types characters such as @ on non-US keyboards
  $mods = [System.Windows.Input.Keyboard]::Modifiers
  $ctrl = (($mods -band [System.Windows.Input.ModifierKeys]::Control) -eq [System.Windows.Input.ModifierKeys]::Control) -and
      (($mods -band [System.Windows.Input.ModifierKeys]::Alt) -ne [System.Windows.Input.ModifierKeys]::Alt)
  $key = [int]$e.Key
  if ($ctrl -and $key -eq [int][System.Windows.Input.Key]::Enter) {
    $index = $script:CurrentPage
    if ($index -ge 0) { Invoke-ButtonClick $PageRunButtons[$index] }
    $e.Handled = $true
  }
  elseif ($key -eq [int][System.Windows.Input.Key]::F5) {
    Invoke-ButtonClick $BtnCases_Refresh
    $e.Handled = $true
  }
  elseif ($ctrl -and $key -ge [int][System.Windows.Input.Key]::D1 -and $key -le [int][System.Windows.Input.Key]::D9) {
    Set-CurrentPage ($key - [int][System.Windows.Input.Key]::D1)
    $e.Handled = $true
  }
  elseif ($ctrl -and $key -eq [int][System.Windows.Input.Key]::D0) {
    Set-CurrentPage 9
    $e.Handled = $true
  }
})

  if($script:SelfTestActive){$e.Cancel=$true;Set-Status 'Self test is running; cleanup must finish before closing.' 'Neutral';return}
  if ($script:Job) {
    $answer = [System.Windows.MessageBox]::Show('An action is still running. Stop it and close?','Action Running','YesNo','Warning')
    if ($answer -ne 'Yes') { $e.Cancel = $true; return }
    Write-Log 'Window closed while an action was running; stopping it.' 'WARN'
    $null = $script:Job.PS.BeginStop($null, $null)
  }
  try { if ($TeamsOutlookTimer) { $TeamsOutlookTimer.Stop() };if($SelfTestReminderTimer){$SelfTestReminderTimer.Stop()} } catch { `
Write-PowerToolsCleanupWarning -Message ('Best-effort operation failed at line {0}: {1}' -f $_.InvocationInfo.ScriptLineNumber, $_.Exception.Message)`
  }
  Save-PowerToolsSetting
})

}
